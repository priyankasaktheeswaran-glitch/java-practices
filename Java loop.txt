import java.util.Scanner;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int n1 = sc.nextInt();
        int n2 = sc.nextInt();
        int n3 = sc.nextInt();

        int digit = 1;

        for (int i = 1; i < n; i++) {
            digit = digit * 10;
        }

        int end = digit * 10 - 1;

        for (int j = digit; j <= end; j++) {

            if (j % n1 == 0 && j % n2 == 0 && j % n3 == 0) {
                System.out.println(j);
                break;
            }
        }
    }
}
