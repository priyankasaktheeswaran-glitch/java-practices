import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        long num = sc.nextLong();
        int digit = sc.nextInt();

        int count = 0;

        while (num > 0) {
            long last = num % 10;

            if (last == digit) {
                count++;
            }

            num = num / 10;
        }

        System.out.println(count);
    }
}
