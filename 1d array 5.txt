import java.util.Scanner;
public class Main{
    public static void main(String[]args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr=new long[n];
        for(int i=0;i<n;i++){
            arr[i]=sc.nextLong();
        }
        long first=sc.nextLong();
        long last=sc.nextLong();
        arr[0]=first;
        arr[n-1]=last;
        for(int i=0;i<n;i++){
            System.out.print(arr[i]+" ");
        }
    }
}
