import java.util.Scanner;
public class Main{
    public static void main(String[]args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for(int i=n-1;i>=0;i--){
            arr[i] = sc.nextLong();
        }
        long num = sc.nextLong();
        long index = -1;
        for(int i=0;i<n;i++){
            if(arr[i]==num){
                index=i;
                break;
            }
        }
        System.out.println(index);
    }
}
